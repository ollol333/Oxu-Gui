local function updatePartButtons()
	HRPBtn.BackgroundColor3 = (aimPartName == "HumanoidRootPart")
		and Color3.fromRGB(80,80,80) or Color3.new(0,0,0)

	HeadBtn.BackgroundColor3 = (aimPartName == "Head")
		and Color3.fromRGB(80,80,80) or Color3.new(0,0,0)
end

	aimPartName = "HumanoidRootPart"
	updatePartButtons()

	aimPartName = "Head"

end)

local function updateButtonColor(player)
	if buttons[player] then
		buttons[player].BackgroundColor3 =
			selectedPlayers[player] and Color3.fromRGB(80,80,80) or Color3.new(0,0,0)
	end
end

local function toggleSelect(player)
	if selectedPlayers[player] then
		selectedPlayers[player] = nil
	else
		selectedPlayers[player] = true
	end
	updateButtonColor(player)
end

--// CREATE BUTTON WITH AVATAR
local function createButton(player)
	if player == LocalPlayer then return end

	local btn = Instance.new("TextButton")
	btn.Size = UDim2.new(1,-10,0,45)
	btn.BackgroundColor3 = Color3.new(0,0,0)
	btn.Text = ""
	btn.Parent = Scroll
	buttons[player] = btn
	Instance.new("UIStroke", btn).Color = Color3.new(1,1,1)

	local avatar = Instance.new("ImageLabel")
	avatar.Size = UDim2.new(0,36,0,36)
	avatar.Position = UDim2.new(0,5,0.5,-18)
	avatar.BackgroundTransparency = 0
	avatar.BackgroundColor3 = Color3.new(0,0,0)
	avatar.Parent = btn
	Instance.new("UIStroke", avatar).Color = Color3.new(1,1,1)

	local img = Players:GetUserThumbnailAsync(
		player.UserId,
		Enum.ThumbnailType.HeadShot,
		Enum.ThumbnailSize.Size420x420
	)
	avatar.Image = img

	local nameLabel = Instance.new("TextLabel")
	nameLabel.Size = UDim2.new(1,-50,1,0)
	nameLabel.Position = UDim2.new(0,50,0,0)
	nameLabel.BackgroundTransparency = 1
	nameLabel.Text = player.Name
	nameLabel.TextColor3 = Color3.new(1,1,1)
	nameLabel.TextScaled = true
	nameLabel.Font = Enum.Font.Gotham
	nameLabel.Parent = btn

	btn.MouseButton1Click:Connect(function()
		toggleSelect(player)
	end)
end

local function removePlayer(player)
	if buttons[player] then
		buttons[player]:Destroy()
		buttons[player] = nil
	end
	selectedPlayers[player] = nil
end

for _,p in pairs(Players:GetPlayers()) do
	createButton(p)
end
Players.PlayerAdded:Connect(createButton)
Players.PlayerRemoving:Connect(removePlayer)

RunService.RenderStepped:Connect(function()
	if not aimEnabled then return end

	local closestPart = nil
	local closestDist = math.huge

	for player in pairs(selectedPlayers) do
		local char = player.Character
		if not char then continue end

		local humanoid = char:FindFirstChildOfClass("Humanoid")
		if not humanoid or humanoid.Health <= 0 then
			-- Está muerto → no se mira
			continue
		end

		local part = char:FindFirstChild(aimPartName)
		if part then
			local dist = (Camera.CFrame.Position - part.Position).Magnitude
			if dist < closestDist then
				closestDist = dist
				closestPart = part
			end
		end
	end

	if closestPart then
		Camera.CFrame = CFrame.new(Camera.CFrame.Position, closestPart.Position)
	end
end)

UserInputService.InputBegan:Connect(function(input, gpe)
	if gpe then return end

	if input.KeyCode == Enum.KeyCode.K then
		Frame.Visible = not Frame.Visible
	end

	if aimKey and input.KeyCode == aimKey then
		aimEnabled = not aimEnabled
	end
end)

ApplyBtn.MouseButton1Click:Connect(function()
	local txt = KeyBox.Text
	if txt ~= "" then
		local key = Enum.KeyCode[txt:upper()]
		if key then
			aimKey = key
		end
	end
end)
